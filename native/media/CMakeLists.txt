cmake_minimum_required(VERSION 3.18)
project(camcat_native_media LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
pybind11_add_module(_camcat_media_native src/sampler.cpp)
target_link_libraries(_camcat_media_native PRIVATE PkgConfig::AV)
target_compile_options(_camcat_media_native PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS _camcat_media_native LIBRARY DESTINATION .)
